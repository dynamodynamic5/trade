  <script>

    function getMaintenanceMarginRate(positionSizeUSD) {

      if (positionSizeUSD <= 50000) return 0.005;

      if (positionSizeUSD <= 200000) return 0.01;

      if (positionSizeUSD <= 500000) return 0.015;

      if (positionSizeUSD <= 1000000) return 0.02;

      return 0.025; // Above 1M USD

    }


    function calculate() {

      const position = document.getElementById('position').value;

      const leverage = parseFloat(document.getElementById('leverage').value);

      const margin = parseFloat(document.getElementById('margin').value);

      const entry = parseFloat(document.getElementById('entry').value);

      const customTarget = parseFloat(document.getElementById('customTarget').value);

      const customType = document.getElementById('customType').value;

      const fee = parseFloat(document.getElementById('fee').value);

      const decimals = parseInt(document.getElementById('decimals').value);


      const targetPercents = [5, 10, 15, 25, 50, 75, 100, 150, 200];

      const results = document.getElementById('results');

      results.innerHTML = "";


      results.innerHTML += `<h2 class='text-xl font-bold'>Profit Target Prices:</h2>`;

      targetPercents.forEach(p => {

        const targetUSD = margin * (p / 100);

        const priceMove = targetUSD / (margin * leverage);

        const priceTarget = position === 'long' ? entry + (entry * priceMove) : entry - (entry * priceMove);

        results.innerHTML += `<div>${p}% → $${priceTarget.toFixed(decimals)}</div>`;

      });


      if (!isNaN(customTarget)) {

        let customMove;

        if (customType === 'percent') {

          customMove = margin * (customTarget / 100);

        } else {

          customMove = customTarget;

        }

        const priceMove = customMove / (margin * leverage);

        const priceTarget = position === 'long' ? entry + (entry * priceMove) : entry - (entry * priceMove);

        results.innerHTML += `<div>Custom Target → $${priceTarget.toFixed(decimals)}</div>`;

      }


      // Position size and maintenance margin based on estimated tier

      const positionSize = (margin * leverage) / entry;

      const positionSizeUSD = positionSize * entry;

      const maintenanceRate = getMaintenanceMarginRate(positionSizeUSD);

      const maintenanceMargin = margin * maintenanceRate;


      let liquidation;

      if (position === 'long') {

        liquidation = (maintenanceMargin - margin + (entry * positionSize)) / positionSize;

      } else {

        liquidation = ((entry * positionSize) - maintenanceMargin + margin) / positionSize;

      }


      results.innerHTML += `<h2 class='text-xl font-bold mt-4'>Liquidation Price:</h2>`;

      results.innerHTML += `<div>$${liquidation.toFixed(decimals)}</div>`;

      results.innerHTML += `<div class='text-sm text-gray-400'>Maintenance Margin Rate Used: ${(maintenanceRate * 100).toFixed(2)}%</div>`;

    }

  </script>
